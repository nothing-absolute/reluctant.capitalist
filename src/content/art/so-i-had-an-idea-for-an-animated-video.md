def generate_video(scene_name, details):
    print(f"Submitting task for: {scene_name}...")
    task = VideoSynthesis.call(
        model="wan2.1-t2v-14b",
        prompt=details["prompt"],
        size=details["size"]
    )
    
    if task.status == 'PENDING':
        print(f"Task ID {task.output.task_id} queued. Waiting for generation...")
        
        while True:
            result = VideoSynthesis.wait(task)
            if result.status == 'SUCCEEDED':
                video_url = result.output.video_url
                print(f"Success! Downloading {scene_name}.mp4")
                urllib.request.urlretrieve(video_url, f"{scene_name}.mp4")
                break
            elif result.status == 'FAILED':
                print(f"Failed: {result.message}")
                break
            
            print("Still processing... checking again in 10 seconds.")
            time.sleep(10)

for name, details in scenes.items():
    generate_video(name, details)
